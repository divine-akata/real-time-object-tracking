starter.py 
import cv2
import numpy as np

cap = cv2.VideoCapture(0)         # open webcam

while True:
    ret, frame = cap.read()         # step 1 — grab frame
    if not ret: break

    hsv = cv2.cvtColor(frame,       # step 2 — BGR → HSV
        cv2.COLOR_BGR2HSV)

    lower_red = np.array([0,120,70]) # step 3 — color mask
    upper_red = np.array([10,255,255])
    mask = cv2.inRange(hsv, lower_red, upper_red)

    contours, _ = cv2.findContours( # step 4 — find edges
        mask, cv2.RETR_TREE,
        cv2.CHAIN_APPROX_SIMPLE)

    for cnt in contours:            # step 5 — draw boxes
        if cv2.contourArea(cnt) > 500:
            x,y,w,h = cv2.boundingRect(cnt)
            cv2.rectangle(frame,(x,y),(x+w,y+h),(0,255,0),2)

    cv2.imshow("Object Tracking", frame)
    if cv2.waitKey(1) & 0xFF == 27: break

cap.release()
cv2.destroyAllWindows()
